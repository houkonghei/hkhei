#include <bits/stdc++.h>
using namespace std;

string sa;
string sb;
vector<int> a;
vector<int> b;
vector<int> c;
vector<char> result;
int carry=0;

int main()
{ cin>>sa>>sb;
  for (int i=sa.size()-1;i>=0;i--)
  { a.push_back(sa[i]-'0');  }
  
  for (int i=sb.size()-1;i>=0;i--)
  { b.push_back(sb[i]-'0');  }
  
  int max_len;
  max_len=sa.size()>sb.size()?sa.size():sb.size();
  int s;
  for (int i=0;i<max_len;i++)
  {   s=carry;
      if (i<sa.size()) {s=s+a[i];}
      if (i<sb.size()) {s=s+b[i];}
      c.push_back(s%10);
      carry=s/10;
  }
  
  if (carry>0)  { c.push_back(carry); }
  
  for (int i=c.size()-1;i>=0;i--)
  {   result.push_back((char)(c[i]+'0'));
  }
  
  for (int i=0;i<result.size();i++)
  {  cout<<result[i];
  }
}
