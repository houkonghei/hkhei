#include<bits/stdc++.h>
using namespace std;
int lena,lenb,lenc;
int a[205]={0},b[205]={0},c[205]={0};
string sa,sb;

void ch(int a[],int b[]){
	int i=1;
	while(i<=lena){
		if(a[i]<b[i]){
			a[i]+=10;
			a[i+1]--;
		}
		c[i]=a[i]-b[i];
		i++;
	}
	lenc=lena;
	while(lenc>1&&c[lenc]==0){
		lenc--;
	}
}

int main(){
	cin>>sa>>sb;
	string t;
	lena=sa.size();
	lenb=sb.size();

	if(lena < lenb || (lena == lenb && sa < sb)){
		t=sa;
		sa=sb;
		sb=t;
		cout<<"-";
	} 	
	lena=sa.size();
	lenb=sb.size();

	for(int i=0;i<lena;i++){
		a[i+1]=sa[lena-i-1]-'0';
	}
	for(int i=0;i<lenb;i++){
		b[i+1]=sb[lenb-i-1]-'0';
	}
	ch(a,b);

	for(int j=lenc;j>0;j--){
		cout<<c[j];
	}
	return 0;
}
